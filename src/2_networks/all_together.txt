import os
from pathlib import Path

import networkx as nx
import pandas as pd
from src.utils import fileio

from settings import main_settings


def create_graph(df_slice, all_nodes):
    G = nx.DiGraph()
    G.add_nodes_from(all_nodes)
    if df_slice.empty:
        return G

    edge_attrs = df_slice.groupby(['node_1', 'node_2'])['duration'].agg(
        ['count', 'sum']).rename(columns={'sum': 'total_interaction_times'})
    for (node_1, node_2), attrs in edge_attrs.iterrows():
        G.add_edge(node_1, node_2, **attrs.to_dict())
    return G


def generate_snapshots(df, all_nodes, window_size, step_size, max_val, output_dir, by_index=False):
    output_dir.mkdir(parents=True, exist_ok=True)
    start, snapshot_idx = 0, 1
    while start < max_val:
        end = start + window_size
        df_slice = df.iloc[int(start):int(end)] if by_index else df[(
            df['start_of_interaction'] >= start) & (df['start_of_interaction'] < end)]

        if not df_slice.empty:
            G = create_graph(df_slice, all_nodes)
            nx.write_gml(G, output_dir / f"{snapshot_idx}.gml")
            snapshot_idx += 1

        start += step_size
        if step_size == 0:
            break


FPS = main_settings.VIDEO_FPS
TREATMENT = main_settings.TREATMENT

TIME_WINDOWS_SEC = [15, 30, 60, 120]
DENSITY_WINDOWS_PERCENT = [0.10, 0.25, 0.33, 0.5]
OVERLAP_PERCENTAGE = 0

INPUT_DIR = Path(main_settings.SAVE_DATA_LOCATION) / \
    "5_social_network_analysis/data/processed" / "2_0_find_interactions" / TREATMENT

BASE_OUTPUT_DIR = Path(main_settings.SAVE_DATA_LOCATION) / "5_social_network_analysis" / "snapshots"

for group_name, group_path in treatment_files.items():
    df = pd.read_csv(group_path, index_col=0).sort_values("start_of_interaction").reset_index(drop=True)
    all_nodes = pd.unique(df[['node_1', 'node_2']].values.ravel('K'))
    overlap_str = f"overlap_{int(OVERLAP_PERCENTAGE*100)}p"

    for window_sec in TIME_WINDOWS_SEC:
        output_dir = BASE_OUTPUT_DIR / overlap_str / \
            f"time_{window_sec}s" / TREATMENT / group_name.replace('.csv', '')
        window = window_sec * FPS
        step = max(1, window * (1 - OVERLAP_PERCENTAGE))
        generate_snapshots(df, all_nodes, window, step, df['start_of_interaction'].max(), output_dir)

    for density_percent in DENSITY_WINDOWS_PERCENT:
        output_dir = BASE_OUTPUT_DIR / overlap_str /\
            f"density_{int(density_percent*100)}p" / TREATMENT / group_name.replace('.csv', '')
        window = int(len(df) * density_percent)
        step = max(1, int(window * (1 - OVERLAP_PERCENTAGE)))
        generate_snapshots(df, all_nodes, window, step, len(df), output_dir, by_index=True)

def rotate_trajectory(df):
    """
    Rotates x,y coordinates by a random angle theta around (0,0).
    Preserves distance from center (Thigmotaxis) but randomizes wall position.
    """
    theta = random.uniform(0, 2 * np.pi)
    c, s = np.cos(theta), np.sin(theta)

    # Assuming data is centered at 0,0.
    # If arena center is (W/2, H/2), subtract center first, rotate, then add back.
    # Here we assume pre-processed data is centered.

    x = df[X_COL].values
    y = df[Y_COL].values

    # Rotation Matrix
    x_new = x * c - y * s
    y_new = x * s + y * c

    df_new = df.copy()
    df_new[X_COL] = x_new
    df_new[Y_COL] = y_new

    return df_new


def process_iteration(iteration_data):
    i, treatment_all, treatment_config, output_dir = iteration_data

    # Create specific temp subfolder for this worker process to avoid collisions
    worker_temp_dir = os.path.join(TEMP_DIR, f"worker_{i}")
    os.makedirs(worker_temp_dir, exist_ok=True)

    # 1. Flatten and Sample
    all_available_flies = []
    for group_name, files_dict in treatment_all.items():
        for fly_name, file_path in files_dict.items():
            all_available_flies.append((fly_name, group_name, file_path))

    selected_flies = random.sample(all_available_flies, FLIES_PER_GROUP)

    pseudo_fly_dict = {}

    # 2. Rotate and Save Temp Files
    for fly_name, group_name, file_path in selected_flies:
        try:
            df = pd.read_csv(file_path)

            df_rotated = rotate_trajectory(df)

            unique_name = f"{fly_name}_{group_name}.csv"
            temp_path = os.path.join(worker_temp_dir, unique_name)
            df_rotated.to_csv(temp_path, index=False)

            unique_key = f"{fly_name}_{group_name}"
            pseudo_fly_dict[unique_key] = temp_path

        except Exception as e:
            print(f"Skipping corrupt file {file_path}: {e}")

    try:
        distances = data_utils.distances_between_all_flies(pseudo_fly_dict)
        angles = data_utils.angles_between_all_flies(pseudo_fly_dict)

        edgelist_df = find_interactions2(angles, distances, treatment_config)

        output_filename = f"pseudo_group_{i}.csv"
        edgelist_df.to_csv(os.path.join(output_dir, output_filename), index=False)

    except Exception as e:
        print(f"Error in iteration {i}: {e}")

    shutil.rmtree(worker_temp_dir, ignore_errors=True)


def find_interactions2(df_angles, df_distances, treatment_config):
    ANGLE = treatment_config["ANGLE"]
    DISTANCE = treatment_config["DISTANCE"]
    max_soc_duration = 1

    edgelist = []
    columns = ["node_1", "node_2", "start_of_interaction", "end_of_interaction", "duration"]

    for angles_col, distances_col in zip(df_angles.columns, df_distances.columns):
        if angles_col != distances_col:
            continue

        dist_vals = df_distances[distances_col].values
        angle_vals = df_angles[angles_col].values

        mask = (dist_vals <= DISTANCE) & (angle_vals >= ANGLE[0]) & (angle_vals <= ANGLE[1])

        if not np.any(mask):
            continue

        padded = np.concatenate(([0], mask.astype(int), [0]))
        diffs = np.diff(padded)
        starts = np.where(diffs == 1)[0]
        ends = np.where(diffs == -1)[0]

        durations = ends - starts
        valid = durations >= max_soc_duration

        starts = starts[valid]
        ends = ends[valid]
        durations = durations[valid]

        if len(starts) == 0:
            continue

        node_1, node_2 = angles_col.split(" ")
        node_1 = node_1.replace(".csv", "")
        node_2 = node_2.replace(".csv", "")

        rows = [[node_1, node_2, int(s), int(e), int(d)] for s, e, d in zip(starts, ends-1, durations)]
        edgelist.extend(rows)

    return pd.DataFrame(edgelist, columns=columns)

    A_time = nx.to_numpy_array(G, weight="total_interaction_times", dtype=float)
    num_nodes = len(G.nodes())

    for g_i in range(1000):
        G_shuffled = nx.DiGraph()
        G_shuffled.add_nodes_from([x for x in range(num_nodes)])
        weights_counts = A_count[A_count != 0].tolist()
        weights_time = A_time[A_time != 0].tolist()
        random.shuffle(weights_counts)
        random.shuffle(weights_time)

        for i in range(num_nodes):
            for j in range(num_nodes):
                if i == j:
                    continue
                elif len(weights_counts) and len(weights_time):
                    G_shuffled.add_edge(i, j, count=weights_counts.pop(), total_interaction_times=weights_time.pop())

        nx.write_gml(G_shuffled, f"{GRAPHS_OUTPUT}/{g_i}.gml")
